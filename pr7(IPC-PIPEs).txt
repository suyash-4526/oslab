#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

#define SIZE 1024

int main()
{
    int pipe1[2], pipe2[2];

    char filename[SIZE];
    char content[SIZE];

    // Create two pipes
    pipe(pipe1);
    pipe(pipe2);

    // Create child process
    pid_t pid = fork();

    if (pid == 0)
    {
        // -------- CHILD PROCESS --------

        // Close unused pipe ends
        close(pipe1[1]);
        close(pipe2[0]);

        // Read filename from parent
        read(pipe1[0], filename, SIZE);

        // Close first pipe after reading
        close(pipe1[0]);

        // Open the requested file
        FILE *fp = fopen(filename, "r");

        if (fp == NULL)
        {
            strcpy(content, "Error: File not found!");
        }
        else
        {
            // Read file contents
            fread(content, 1, SIZE - 1, fp);
            fclose(fp);
        }

        // Send file contents to parent
        write(pipe2[1], content, strlen(content) + 1);

        // Close second pipe
        close(pipe2[1]);

        exit(0);
    }
    else if (pid > 0)
    {
        // -------- PARENT PROCESS --------

        // Close unused pipe ends
        close(pipe1[0]);
        close(pipe2[1]);

        // Accept filename from user
        printf("Enter file name: ");
        scanf("%s", filename);

        // Send filename to child
        write(pipe1[1], filename, strlen(filename) + 1);

        // Close first pipe after writing
        close(pipe1[1]);

        // Read file contents from child
        read(pipe2[0], content, SIZE);

        // Display file contents
        printf("\nContents of file:\n");
        printf("%s\n", content);

        // Close second pipe
        close(pipe2[0]);

        // Wait for child process
        wait(NULL);
    }
    else
    {
        perror("fork failed");
        return 1;
    }

    return 0;
}
