#include <stdio.h>

int main()
{
    int n, m;

    // Input number of processes and resources
    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter number of resources: ");
    scanf("%d", &m);

    int allocation[n][m], max[n][m];
    int need[n][m], available[m];

    int finish[n], safeSequence[n];

    // Input Allocation Matrix
    printf("\nEnter Allocation Matrix:\n");
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            scanf("%d", &allocation[i][j]);
        }
    }

    // Input Maximum Matrix
    printf("\nEnter Max Matrix:\n");
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            scanf("%d", &max[i][j]);
        }
    }

    // Input Available Resources
    printf("\nEnter Available Resources:\n");
    for (int j = 0; j < m; j++)
    {
        scanf("%d", &available[j]);
    }

    // Calculate Need Matrix
    for (int i = 0; i < n; i++)
    {
        finish[i] = 0;  // Initially, no process is finished

        for (int j = 0; j < m; j++)
        {
            need[i][j] = max[i][j] - allocation[i][j];
        }
    }

    // Banker's Safety Algorithm
    int count = 0;

    while (count < n)
    {
        int found = 0;

        for (int i = 0; i < n; i++)
        {
            // Check if process is unfinished
            if (finish[i] == 0)
            {
                int j;

                // Check whether available resources satisfy its need
                for (j = 0; j < m; j++)
                {
                    if (need[i][j] > available[j])
                        break;
                }

                // If all resources are available
                if (j == m)
                {
                    // Process completes and releases its resources
                    for (int k = 0; k < m; k++)
                    {
                        available[k] += allocation[i][k];
                    }

                    safeSequence[count] = i;
                    count++;

                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        // If no process can execute, system is unsafe
        if (found == 0)
            break;
    }

    // Display result
    if (count == n)
    {
        printf("\nSystem is in SAFE STATE\n");
        printf("Safe Sequence: ");

        for (int i = 0; i < n; i++)
        {
            printf("P%d", safeSequence[i]);

            if (i != n - 1)
                printf(" -> ");
        }

        printf("\n");
    }
    else
    {
        printf("\nSystem is in UNSAFE STATE\n");
        printf("No safe sequence exists.\n");
    }

    return 0;
}
