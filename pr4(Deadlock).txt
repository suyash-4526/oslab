#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define N 5

// Semaphore for each fork
sem_t fork[N];

// Counting semaphore to allow only 4 philosophers at a time
sem_t room;

// Philosopher function
void *philosopher(void *arg)
{
    int id = *(int *)arg;

    // Repeat eating 2 times
    for (int i = 0; i < 2; i++)
    {
        printf("Philosopher %d is thinking\n", id);

        // Allow only 4 philosophers to enter
        sem_wait(&room);

        // Pick up left fork
        sem_wait(&fork[id]);

        // Pick up right fork
        sem_wait(&fork[(id + 1) % N]);

        // Eat
        printf("Philosopher %d is eating\n", id);
        sleep(1);

        // Put down left fork
        sem_post(&fork[id]);

        // Put down right fork
        sem_post(&fork[(id + 1) % N]);

        // Leave the room
        sem_post(&room);

        printf("Philosopher %d finished eating\n", id);
    }

    return NULL;
}

int main()
{
    pthread_t t[N];
    int id[N];

    // Initialize all fork semaphores to 1
    for (int i = 0; i < N; i++)
        sem_init(&fork[i], 0, 1);

    // Initialize room semaphore to 4
    sem_init(&room, 0, N - 1);

    // Create 5 philosopher threads
    for (int i = 0; i < N; i++)
    {
        id[i] = i;
        pthread_create(&t[i], NULL, philosopher, &id[i]);
    }

    // Wait for all philosophers to finish
    for (int i = 0; i < N; i++)
        pthread_join(t[i], NULL);

    // Destroy semaphores
    for (int i = 0; i < N; i++)
        sem_destroy(&fork[i]);

    sem_destroy(&room);

    return 0;
}
